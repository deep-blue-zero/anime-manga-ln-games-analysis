from pathlib import PurePosixPath, Path
parser = argparse.ArgumentParser()
parser.add_argument('archive')
args = parser.parse_args()
archive = Path(args.archive)
with zipfile.ZipFile(archive) as z:
    assert z.testzip() is None, 'ZIP CRC failure'
    names = [i.filename for i in z.infolist() if not i.is_dir()]
    assert len(names) == len(set(names)), 'Duplicate ZIP members'
    assert all(PurePosixPath(n).suffix.lower() == '.md' for n in names)
    assert all(not PurePosixPath(n).is_absolute() and '..' not in PurePosixPath(n).parts for n in names)
    roots = {PurePosixPath(n).parts[0] for n in names}
    assert len(roots) == 1
    root = next(iter(roots)) + '/'
    assert root[:-1] in {'07_SHINOSAWA_HIRO', '11_HATAYA_MISUZU'}
    checksum_name = root + 'CHECKSUMS.md'
    text = z.read(checksum_name).decode('utf-8-sig')
    rows = [line.strip().strip('|').split('|') for line in text.splitlines() if line.startswith('|')]
    inventory = {r[0].strip(): r[1].strip() for r in rows if len(r) == 2 and re.fullmatch('[a-f0-9]{64}', r[1].strip())}
    assert set(inventory) == {n[len(root):] for n in names if n != checksum_name}
    for name, expected in inventory.items():

print('ZIP SHA-256:', hashlib.sha256(archive.read_bytes()).hexdigest())
