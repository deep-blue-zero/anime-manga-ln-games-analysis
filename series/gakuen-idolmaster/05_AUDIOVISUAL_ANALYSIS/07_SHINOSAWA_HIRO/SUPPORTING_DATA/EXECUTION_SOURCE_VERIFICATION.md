### Exact packet comparison code

Save the following fenced code as `compare_h04.py` in a working directory. Python 3.12 and FFprobe 8.1.1 were used. Supply the full original and prior trim explicitly; the script verifies both required SHA-256 values before probing. No separate manifest or working JSON file is required. The optional output JSON is a working result, not an additional packaged artifact.

```powershell
python compare_h04.py --original "<full H04 original path>" --trim "<prior H04 trim path>" --output "h04_packet_comparison.json"
```

```python
import pathlib,json,subprocess,hashlib,concurrent.futures,argparse
parser=argparse.ArgumentParser(description='Compare exact hash-verified H04 trim packets with the original prefix.')
parser.add_argument('--original',required=True)
parser.add_argument('--trim',required=True)
parser.add_argument('--output')
args=parser.parse_args()
def identity(path,expected):
    source=pathlib.Path(path).resolve(strict=True)
    if not source.is_file():raise ValueError('Input is not a file: '+str(source))
    h=hashlib.sha256()
    with source.open('rb') as f:
        for block in iter(lambda:f.read(8*1024*1024),b''):h.update(block)
    actual=h.hexdigest()
    if actual!=expected:raise ValueError('Input hash mismatch: '+str(source))
    return {'path':str(source),'sha256':actual}
orig=identity(args.original,'4dc80d8f2431fe8780faf59fa6b9f372f0fafcb599607ce9e04ead81bc4d4468')
trim=identity(args.trim,'de92afafb3181342407ab9405f4951e820bfdafaad5b1e9538ec9d7f73ea826e')
def packets(r,kind):
    cmd=['ffprobe','-v','error','-select_streams',kind+':0','-show_packets','-show_entries','packet=pts,dts,duration,size,data_hash','-show_data_hash','sha256','-of','json',r['path']]
    p=subprocess.run(cmd,capture_output=True,check=True)
    return json.loads(p.stdout)['packets']
def digest(seq):return hashlib.sha256(('\n'.join(p['data_hash'] for p in seq)+'\n').encode()).hexdigest()
out={'method':'ffprobe ordered compressed packet payload SHA256, comparing trim to original prefix; no video/audio re-encode','original_sha256':orig['sha256'],'trim_sha256':trim['sha256'],'streams':{}}
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    for kind in ['v','a']:
        a,b=list(pool.map(lambda r:packets(r,kind),[orig,trim]))
        n=min(len(a),len(b));mismatches=[i for i in range(n) if a[i]['data_hash']!=b[i]['data_hash']]
        pts_diff=[b[i].get('pts',0)-a[i].get('pts',0) for i in range(n)]
        dts_diff=[b[i].get('dts',0)-a[i].get('dts',0) for i in range(n)]
        duration_diff=[i for i in range(n) if a[i].get('duration')!=b[i].get('duration')]
        out['streams'][kind]={'original_packets':len(a),'trim_packets':len(b),'compared_packets':n,'payload_mismatch_count':len(mismatches),'first_payload_mismatches':mismatches[:10],'all_trim_packets_match_original_prefix':len(b)<=len(a) and not mismatches,'pts_offset_min':min(pts_diff),'pts_offset_max':max(pts_diff),'dts_offset_min':min(dts_diff),'dts_offset_max':max(dts_diff),'duration_difference_count':len(duration_diff),'first_duration_difference_indices':duration_diff[:10],'original_prefix_payload_digest':digest(a[:n]),'trim_payload_digest':digest(b),'first_original_packet':a[0],'first_trim_packet':b[0],'last_compared_original_packet':a[n-1],'last_trim_packet':b[-1],'first_extra_original_packet':a[n] if n<len(a) else None}
        print(kind,out['streams'][kind]['all_trim_packets_match_original_prefix'],n,flush=True)
if args.output:pathlib.Path(args.output).write_text(json.dumps(out,indent=2),encoding='utf-8')
print(json.dumps(out,indent=2))
```

All packet-digest comparisons are computed from the verified input files, not from the recorded conclusion.
